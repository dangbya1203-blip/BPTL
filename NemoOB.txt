-- ===================== UTILS =====================
local function getRootPart(char)
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function getHumanoid(char)
    return char and char:FindFirstChildOfClass("Humanoid")
end

local function isTeammate(player)
    if not Config.TeamCheck then return false end
    local lp = Players.LocalPlayer
    return lp and lp.Team ~= nil and player.Team == lp.Team
end

-- Line of Sight Raycast check
local function isVisible(targetPart, targetChar)
    local camera = workspace.CurrentCamera
    if not camera then return false end

    local origin = camera.CFrame.Position
    local destination = targetPart.Position
    local direction = destination - origin

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude

    local excludeList = {}
    local lp = Players.LocalPlayer
    if lp and lp.Character then
        table.insert(excludeList, lp.Character)
    end
    if targetChar then
        table.insert(excludeList, targetChar)
    end
    params.FilterDescendantsInstances = excludeList

    local result = workspace:Raycast(origin, direction, params)
    return result == nil
end

local function getBoundingBox(char)
    local root = getRootPart(char)
    if not root then return nil end

    local camera = workspace.CurrentCamera
    if not camera then return nil end

    -- Fast pre-check: if root is not in viewport or behind the camera, skip calculations
    local _, inViewport = camera:WorldToViewportPoint(root.Position)
    if not inViewport then return nil end

    local cf = root.CFrame
    local width  = 4
    local height = 5.5

    local corners = {
        cf * CFrame.new(-width/2,  height/2, 0),
        cf * CFrame.new( width/2,  height/2, 0),
        cf * CFrame.new(-width/2, -height/2, 0),
        cf * CFrame.new( width/2, -height/2, 0),
    }

    local minX, minY = math.huge, math.huge
    local maxX, maxY = -math.huge, -math.huge
    local visible = false

    for i = 1, 4 do
        local screenPos, inViewport = camera:WorldToViewportPoint(corners[i].Position)
        if screenPos.Z > 0 then
            visible = true
            if screenPos.X < minX then minX = screenPos.X end
            if screenPos.Y < minY then minY = screenPos.Y end
            if screenPos.X > maxX then maxX = screenPos.X end
            if screenPos.Y > maxY then maxY = screenPos.Y end
        end
    end

    if not visible then return nil end

    return {
        x  = minX,
        y  = minY,
        w  = maxX - minX,
        h  = maxY - minY,
        cx = (minX + maxX) / 2,
        cy = (minY + maxY) / 2,
    }
end

-- ===================== RENDER UPDATES =====================
local function updateCornerBox(lines, box, color, thick, ratio)
    if not box then
        for i = 1, 8 do lines[i].Visible = false end
        return
    end
    local x, y, w, h = box.x, box.y, box.w, box.h
    local lx = w * ratio
    local ly = h * ratio
    local corners = {
        {Vector2.new(x,   y),   Vector2.new(x+lx,   y),   Vector2.new(x,   y+ly)  },
        {Vector2.new(x+w, y),   Vector2.new(x+w-lx, y),   Vector2.new(x+w, y+ly)  },
        {Vector2.new(x,   y+h), Vector2.new(x+lx,   y+h), Vector2.new(x,   y+h-ly)},
        {Vector2.new(x+w, y+h), Vector2.new(x+w-lx, y+h), Vector2.new(x+w, y+h-ly)},
    }
    for i, c in ipairs(corners) do
        local hl = lines[(i-1)*2 + 1]
        local vl = lines[(i-1)*2 + 2]
        hl.From = c[1]; hl.To = c[2]; hl.Color = color; hl.Thickness = thick; hl.Visible = true
        vl.From = c[1]; vl.To = c[3]; vl.Color = color; vl.Thickness = thick; vl.Visible = true
    end
end

local function updateHealthBar(bg, fg, box, hp, maxHp)
    if not box then
        bg.Visible = false
        fg.Visible = false
        return
    end
    local bx = box.x - 5
    local ratio = math.clamp(hp / math.max(maxHp, 1), 0, 1)
    bg.From      = Vector2.new(bx, box.y)
    bg.To        = Vector2.new(bx, box.y + box.h)
    bg.Color     = Color3.fromRGB(0, 0, 0)
    bg.Thickness = Config.HealthBarWidth + 2
    bg.Visible   = true
    fg.From      = Vector2.new(bx, box.y + box.h)
    fg.To        = Vector2.new(bx, box.y + box.h - box.h * ratio)
    fg.Color     = Color3.fromRGB(math.floor(255*(1-ratio)), math.floor(255*ratio), 40)
    fg.Thickness = Config.HealthBarWidth
    fg.Visible   = true
end

local function getTracerOrigin()
    local camera = workspace.CurrentCamera
    local vp = camera and camera.ViewportSize or Vector2.new(800, 600)
    if Config.TracerOrigin == "Top"    then return Vector2.new(vp.X/2, 0)      end
    if Config.TracerOrigin == "Center" then return Vector2.new(vp.X/2, vp.Y/2) end
    return Vector2.new(vp.X/2, vp.Y)
end

local function updateTracer(line, box, color, thick)
    if not box then line.Visible = false; return end
    line.From      = getTracerOrigin()
    line.To        = Vector2.new(box.cx, box.y + box.h)
    line.Color     = color
    line.Thickness = thick
    line.Visible   = true
end

local function worldToMap(worldPos)
    local lp = Players.LocalPlayer
    local lChar = lp and lp.Character
    if not lChar then return nil end
    local lRoot = getRootPart(lChar)
    if not lRoot then return nil end

    local camera = workspace.CurrentCamera
    if not camera then return nil end

    local off   = worldPos - lRoot.Position
    local half  = MapCfg.MapSize / 2
    local scale = half / MapCfg.WorldRange
    local look  = camera.CFrame.LookVector
    local angle = math.atan2(look.X, look.Z)
    local ca    = math.cos(-angle)
    local sa    = math.sin(-angle)
    local rx    = off.X * ca - off.Z * sa
    local rz    = off.X * sa + off.Z * ca
    local sx    = math.clamp(MapCfg.PosX + half + rx * scale, MapCfg.PosX + 2, MapCfg.PosX + MapCfg.MapSize - 2)
    local sy    = math.clamp(MapCfg.PosY + half + rz * scale, MapCfg.PosY + 2, MapCfg.PosY + MapCfg.MapSize - 2)
    return Vector2.new(sx, sy)
end

local function updateMinimap()
    local show = Config.TopDown and Config.Enabled
    MapUI.bg.Visible      = show
    MapUI.border.Visible  = show
    MapUI.selfDot.Visible = show
    if not show then
        for _, dot in pairs(MapDots) do dot.Visible = false end
        return
    end
    MapUI.selfDot.Position = Vector2.new(MapCfg.PosX + MapCfg.MapSize/2, MapCfg.PosY + MapCfg.MapSize/2)
    for player, dot in pairs(MapDots) do
        if not player or not player.Parent then
            dot:Remove()
            MapDots[player] = nil
        elseif isTeammate(player) then
            dot.Visible = false
        else
            local char = player.Character
            local root = char and getRootPart(char)
            if root then
                local mp = worldToMap(root.Position)
                if mp then
                    dot.Position = mp
                    dot.Color    = Config.TopDownColor
                    dot.Visible  = true
                else
                    dot.Visible = false
                end
            else
                dot.Visible = false
            end
        end
    end
end

-- ===================== CHAMS (WALLHACK HIGHLIGHT) =====================
local function applyChams(char, obj)
    if not Config.Chams then
        if obj.highlight then
            obj.highlight:Destroy()
            obj.highlight = nil
        end
        return
    end

    if not obj.highlight or obj.highlight.Parent ~= char then
        if obj.highlight then obj.highlight:Destroy() end
        local hl = Instance.new("Highlight")
        hl.Name = "PremiumCham"
        hl.Adornee = char
        hl.Parent = char
        obj.highlight = hl
    end

    local hl = obj.highlight
    hl.FillColor = Config.ChamsFillColor
    hl.OutlineColor = Config.ChamsOutlineColor
    hl.FillTransparency = Config.ChamsFillTrans
    hl.OutlineTransparency = Config.ChamsOutlineTrans
    hl.Enabled = Config.Enabled
end

-- ===================== AIMBOT ENGINE =====================
local function isKeyHeld(key)
    if UserInputService.TouchEnabled and isMobileAimbotPressed then
        return true
    end
    if key == Enum.UserInputType.MouseButton1 or key == Enum.UserInputType.MouseButton2 or key == Enum.UserInputType.MouseButton3 then
        return UserInputService:IsMouseButtonPressed(key)
    else
        return UserInputService:IsKeyDown(key)
    end
end

local function getClosestPlayer()
    local lp = Players.LocalPlayer
    local lChar = lp and lp.Character
    local lRoot = lChar and getRootPart(lChar)
    local camera = workspace.CurrentCamera
    if not lp or not lRoot or not camera then return nil end

    local target = nil
    local minDistance3D = math.huge
    local fovRadius = Config.AimbotFOVRadius

    local vp = camera.ViewportSize
    local screenCenter = Vector2.new(vp.X / 2, vp.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= lp and not isTeammate(player) then
            local char = player.Character
            if char then
                local hum = getHumanoid(char)
                if hum and hum.Health > 0 then
                    local targetPart = char:FindFirstChild(Config.AimbotPart)
                    if targetPart then
                        local screenPos, inViewport = camera:WorldToViewportPoint(targetPart.Position)
                        if screenPos.Z > 0 then
                            local inFOV = true
                            if Config.AimbotFOV then
                                local distFromCenter = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                                if distFromCenter > fovRadius then
                                    inFOV = false
                                end
                            end

                            if inFOV then
                                local visible = true
                                if Config.WallCheck then
                                    visible = isVisible(targetPart, char)
                                end

                                if visible then
                                    local dist3D = (targetPart.Position - lRoot.Position).Magnitude
                                    if dist3D < minDistance3D then
                                        minDistance3D = dist3D
                                        target = targetPart
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return target
end

-- Frame-Rate Independent smoothing formula
local function runAimbot(dt)
    if not Config.Aimbot or not isKeyHeld(Config.AimbotKey) then
        currentTarget = nil
        return
    end

    local camera = workspace.CurrentCamera
    if not camera then return end

    local lp = Players.LocalPlayer
    local lChar = lp and lp.Character
    local lRoot = lChar and getRootPart(lChar)
    if not lRoot then return end

    local targetValid = false
    if currentTarget and currentTarget.Parent and currentTarget.Parent:FindFirstChildOfClass("Humanoid") then
        local hum = currentTarget.Parent:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health > 0 then
            local screenPos, inViewport = camera:WorldToViewportPoint(currentTarget.Position)
            if screenPos.Z > 0 then
                local inFOV = true
                if Config.AimbotFOV then
                    local vp = camera.ViewportSize
                    local screenCenter = Vector2.new(vp.X / 2, vp.Y / 2)
                    local distFromCenter = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
                    if distFromCenter > Config.AimbotFOVRadius then
                        inFOV = false
                    end
                end

                local visible = true
                if Config.WallCheck then
                    visible = isVisible(currentTarget, currentTarget.Parent)
                end

                if inFOV and visible then
                    targetValid = true
                end
            end
        end
    end

    if not targetValid then
        currentTarget = getClosestPlayer()
    end

    if currentTarget then
        local currentCF = camera.CFrame
        local targetPos = currentTarget.Position

        if Config.AimbotPrediction > 0 then
            local velocity = currentTarget.AssemblyLinearVelocity or currentTarget.Velocity or Vector3.new(0, 0, 0)
            targetPos = targetPos + (velocity * Config.AimbotPrediction)
        end

        local targetCF = CFrame.new(currentCF.Position, targetPos)

        if Config.AimbotSmooth > 0 then
            -- Compensate smoothing for 60Hz/144Hz/240Hz monitors to keep speed consistent
            local smoothFactor = math.clamp(Config.AimbotSmooth * (dt * 60), 0.001, 1)
            camera.CFrame = currentCF:Lerp(targetCF, smoothFactor)
        else
            camera.CFrame = targetCF
        end
    end
end

-- ===================== HITBOX EXPANDER =====================
local function expandHitboxes()
    if not Config.HitboxExpand or not Config.Enabled then
        restoreHitboxes()
        return
    end

    local lp = Players.LocalPlayer
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= lp and not isTeammate(player) then
            local char = player.Character
            if char then
                local part = char:FindFirstChild(Config.HitboxPartToSize)
                if part and part:IsA("BasePart") then
                    if not originalProperties[part] then
                        originalProperties[part] = {
                            Size = part.Size,
                            Transparency = part.Transparency,
                            CanCollide = part.CanCollide
                        }
                    end
                    part.Size = Vector3.new(Config.HitboxSize, Config.HitboxSize, Config.HitboxSize)
                    part.Transparency = Config.HitboxTrans
                    part.CanCollide = false
                end
            end
        end
    end
end

local function restoreHitboxes()
    for part, props in pairs(originalProperties) do
        if part and part.Parent then
            part.Size = props.Size
            part.Transparency = props.Transparency
            part.CanCollide = props.CanCollide
        end
    end
    originalProperties = {}
end

-- ===================== ESP CREATOR / DESTROYER =====================
local function createESP(player)
    local lines = {}
    for i = 1, 8 do lines[i] = newLine() end
    ESPObjects[player] = {
        cornerLines = lines,
        tracer      = newLine(),
        hpBg        = newLine(),
        hpFg        = newLine(),
        nameText    = newText(),
        distText    = newText(),
        highlight   = nil,
    }
    local dot = newCircle()
    dot.Color   = Config.TopDownColor
    dot.Radius  = MapCfg.DotRadius
    dot.ZIndex  = 6
    MapDots[player] = dot
end

local function hideESP(obj)
    for i = 1, 8 do obj.cornerLines[i].Visible = false end
    obj.tracer.Visible   = false
    obj.hpBg.Visible     = false
    obj.hpFg.Visible     = false
    obj.nameText.Visible = false
    obj.distText.Visible = false
    if obj.highlight then
        obj.highlight.Enabled = false
    end
end

local function removeESP(player)
    local obj = ESPObjects[player]
    if obj then
        for i = 1, 8 do obj.cornerLines[i]:Remove() end
        obj.tracer:Remove()
        obj.hpBg:Remove()
        obj.hpFg:Remove()
        obj.nameText:Remove()
        obj.distText:Remove()
        if obj.highlight then obj.highlight:Destroy() end
        ESPObjects[player] = nil
    end
    local dot = MapDots[player]
    if dot then dot:Remove(); MapDots[player] = nil end
end

-- ===================== UPDATE PER PLAYER =====================
local function updatePlayer(player, obj)
    if isTeammate(player) then hideESP(obj); return end

    local char = player.Character
    if not char then hideESP(obj); return end

    local root = getRootPart(char)
    if not root then hideESP(obj); return end

    local lp = Players.LocalPlayer
    local lChar = lp and lp.Character
    local lRoot = lChar and getRootPart(lChar)
    local dist  = lRoot and (root.Position - lRoot.Position).Magnitude or 0
    if dist > Config.MaxDist then hideESP(obj); return end

    local hp, maxHp = 100, 100
    local hum = getHumanoid(char)
    if hum then hp = hum.Health; maxHp = hum.MaxHealth end

    -- Render Chams (Wallhack Highlight Glow)
    applyChams(char, obj)

    -- Render 2D Bounding Box
    local box = getBoundingBox(char)

    if Config.CornerBox then
        updateCornerBox(obj.cornerLines, box, Config.CornerBoxColor, Config.CornerBoxThick, Config.CornerBoxLen)
    else
        for i = 1, 8 do obj.cornerLines[i].Visible = false end
    end

    if Config.HealthBar then
        updateHealthBar(obj.hpBg, obj.hpFg, box, hp, maxHp)
    else
        obj.hpBg.Visible = false; obj.hpFg.Visible = false
    end

    if Config.Tracer then
        updateTracer(obj.tracer, box, Config.TracerColor, Config.TracerThick)
    else
        obj.tracer.Visible = false
    end

    if Config.ShowName and box then
        obj.nameText.Text     = player.DisplayName
        obj.nameText.Position = Vector2.new(box.cx, box.y - 18)
        obj.nameText.Color    = Config.NameColor
        obj.nameText.Size     = Config.NameSize
        obj.nameText.Visible  = true
    else
        obj.nameText.Visible = false
    end

    if Config.ShowDist and box then
        obj.distText.Text     = "[" .. math.floor(dist) .. "m]"
        obj.distText.Position = Vector2.new(box.cx, box.y + box.h + 3)
        obj.distText.Color    = Config.DistColor
        obj.distText.Size     = Config.DistSize
        obj.distText.Visible  = true
    else
        obj.distText.Visible = false
    end
end

-- ===================== CORE RENDER LOOP =====================
local function onRenderStep(dt)
    -- Update FOV circle
    local showFOV = Config.Enabled and Config.Aimbot and Config.AimbotFOV
    FOVCircle.Visible = showFOV
    if showFOV then
        local camera = workspace.CurrentCamera
        if camera then
            local vp = camera.ViewportSize
            FOVCircle.Position  = Vector2.new(vp.X / 2, vp.Y / 2)
            FOVCircle.Radius    = Config.AimbotFOVRadius
            FOVCircle.Color     = Config.AimbotFOVColor
            FOVCircle.Thickness = Config.AimbotFOVThick
            FOVCircle.Filled    = Config.AimbotFOVFilled
        end
    end

    if not Config.Enabled then
        for _, obj in pairs(ESPObjects) do hideESP(obj) end
        restoreHitboxes()
        updateMinimap()
        return
    end

    -- Run Rainbow mode (RGB Color Cycle)
    if Config.Rainbow then
        local hue = (tick() % 5) / 5
        local color = Color3.fromHSV(hue, 0.8, 1)
        Config.CornerBoxColor = color
        Config.TracerColor = color
        Config.ChamsFillColor = color
        Config.TopDownColor = color
        Config.AimbotFOVColor = color
    end

    -- Run Aimbot
    local okAim, errAim = pcall(runAimbot, dt)
    if not okAim then
        warn("[Aimbot Error] " .. tostring(errAim))
    end

    -- Find Silent Aim target if aimbot key is not held
    if Config.SilentAim and not isKeyHeld(Config.AimbotKey) then
        currentTarget = getClosestPlayer()
    end

    -- Run Hitbox Expander
    local okHitbox, errHitbox = pcall(expandHitboxes)
    if not okHitbox then
        warn("[Hitbox Error] " .. tostring(errHitbox))
    end

    -- Update ESP Visuals
    for player, obj in pairs(ESPObjects) do
        if player and player.Parent then
            local ok, err = pcall(updatePlayer, player, obj)
            if not ok then 
                warn("[ESP Error] " .. tostring(err))
                hideESP(obj) 
            end

local function makeDraggable(frame, dragHandle)
    local dragToggle = nil
    local dragInput = nil
    local dragStart = nil
    local dragStartPos = nil

    local function updateInput(input)
        local delta = input.Position - dragStart
        local position = UDim2.new(dragStartPos.X.Scale, dragStartPos.X.Offset + delta.X, dragStartPos.Y.Scale, dragStartPos.Y.Offset + delta.Y)
        TweenService:Create(frame, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Position = position}):Play()
    end

    local handle = dragHandle or frame
    local conn1 = handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragToggle = true
            dragStart = input.Position
            dragStartPos = frame.Position
        end
    end)
    table.insert(uiConnections, conn1)

    local conn2 = handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    table.insert(uiConnections, conn2)

    local conn3 = UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragToggle = false
        end
    end)
    table.insert(uiConnections, conn3)

    local conn4 = UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragToggle then
            updateInput(input)
        end
    end)
    table.insert(uiConnections, conn4)
end

local function CalculateChance(Percentage)
    Percentage = math.floor(Percentage)
    local chance = math.floor(Random.new().NextNumber(Random.new(), 0, 1) * 100) / 100
    return chance <= Percentage / 100
end

local function ValidateArguments(Args, RayMethod)
    local Matches = 0
    if #Args < RayMethod.ArgCountRequired then
        return false
    end
    for Pos, Argument in next, Args do
        if typeof(Argument) == RayMethod.Args[Pos] then
            Matches = Matches + 1
        end
    end
    return Matches >= RayMethod.ArgCountRequired
end

    return (Position - Origin).Unit * 1000

        local lp = Players.LocalPlayer
        if not lp then return end
        local mouse = lp:GetMouse()
        local mt = getrawmetatable(game)
        local oldIndex = mt.__index
        local oldNamecall
        
        local isWritable = setreadonly or make_writeable
        if not isWritable then return end
        
        if setreadonly then setreadonly(mt, false) else make_writeable(mt) end

        mt.__index = newcclosure(function(self, Index)
            if Config.Enabled and Config.SilentAim and Config.SilentAimMethod == "Mouse.Hit/Target" and currentTarget and self == mouse and not checkcaller() then
                local chance = CalculateChance(Config.SilentAimHitChance)
                if chance then
                    if Index == "Target" or Index == "target" then 
                        return currentTarget
                    elseif Index == "Hit" or Index == "hit" then 
                        local velocity = currentTarget.AssemblyLinearVelocity or currentTarget.Velocity or Vector3.new(0, 0, 0)
                        return ((Config.AimbotPrediction > 0 and (currentTarget.CFrame + (velocity * Config.AimbotPrediction))) or currentTarget.CFrame)
                    elseif Index == "X" or Index == "x" then 
                        return self.X 
                    elseif Index == "Y" or Index == "y" then 
                        return self.Y 
                    elseif Index == "UnitRay" then 
                        local origin = self.Origin.Position
                        local targetPos = currentTarget.Position
                        if Config.AimbotPrediction > 0 then
                            local velocity = currentTarget.AssemblyLinearVelocity or currentTarget.Velocity or Vector3.new(0, 0, 0)
                            targetPos = targetPos + (velocity * Config.AimbotPrediction)
                        end
                        return Ray.new(origin, (targetPos - origin).Unit)
                    end
                end
            end
            return oldIndex(self, Index)
        end)

        oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
            local Method = getnamecallmethod()
            local Arguments = {...}
            local self = Arguments[1]
            
            if Config.Enabled and Config.SilentAim and self == workspace and not checkcaller() and currentTarget then
                if Method == "FindPartOnRayWithIgnoreList" and Config.SilentAimMethod == Method then
                    if ValidateArguments(Arguments, ExpectedArguments.FindPartOnRayWithIgnoreList) then
                        local chance = CalculateChance(Config.SilentAimHitChance)
                        if chance then
                            local A_Ray = Arguments[2]
                            local Origin = A_Ray.Origin
                            local Direction = getDirection(Origin, currentTarget.Position)
                            Arguments[2] = Ray.new(Origin, Direction)
                            return oldNamecall(unpack(Arguments))
                        end
                    end
                elseif Method == "FindPartOnRayWithWhitelist" and Config.SilentAimMethod == Method then
                    if ValidateArguments(Arguments, ExpectedArguments.FindPartOnRayWithWhitelist) then
                        local chance = CalculateChance(Config.SilentAimHitChance)
                        if chance then
                            local A_Ray = Arguments[2]
                            local Origin = A_Ray.Origin
                            local Direction = getDirection(Origin, currentTarget.Position)
                            Arguments[2] = Ray.new(Origin, Direction)
                            return oldNamecall(unpack(Arguments))
                        end
                    end
                elseif (Method == "FindPartOnRay" or Method == "findPartOnRay") and Config.SilentAimMethod:lower() == Method:lower() then
                    if ValidateArguments(Arguments, ExpectedArguments.FindPartOnRay) then
                        local chance = CalculateChance(Config.SilentAimHitChance)
                        if chance then
                            local A_Ray = Arguments[2]
                            local Origin = A_Ray.Origin
                            local Direction = getDirection(Origin, currentTarget.Position)
                            Arguments[2] = Ray.new(Origin, Direction)
                            return oldNamecall(unpack(Arguments))
                        end
                    end
                elseif Method == "Raycast" and Config.SilentAimMethod == Method then
                    if ValidateArguments(Arguments, ExpectedArguments.Raycast) then
                        local chance = CalculateChance(Config.SilentAimHitChance)
                        if chance then
                            local A_Origin = Arguments[2]
                            Arguments[3] = getDirection(A_Origin, currentTarget.Position)
                            return oldNamecall(unpack(Arguments))
                        end
                    end
                end
            end
            return oldNamecall(...)
        end))

        if setreadonly then setreadonly(mt, true) else make_readonly(mt) end
    end)
    if not success then
        warn("[Silent Aim Warning] Metatable hooking is not supported: " .. tostring(err))
    end
end

-- ===================== EXPORT GLOBAL API =====================
local ESP = {}

function ESP.Unload()
    pcall(function() RunService:UnbindFromRenderStep("UltimateESP_RenderLoop") end)
    for player in pairs(ESPObjects) do removeESP(player) end
    restoreHitboxes()
    if MapUI.bg      then MapUI.bg:Remove()      end
    if MapUI.border  then MapUI.border:Remove()  end
    if MapUI.selfDot then MapUI.selfDot:Remove() end
    if FOVCircle     then FOVCircle:Remove()     end
    
    -- Cleanup Mobile GUI
    for _, conn in ipairs(uiConnections) do
        if conn and conn.Disconnect then
            pcall(function() conn:Disconnect() end)
        end
    end
    uiConnections = {}
    if MobileUIObj.screenGui then
        pcall(function() MobileUIObj.screenGui:Destroy() end)
        MobileUIObj.screenGui = nil
    end
    
    print("[ESP/Aimbot] Fully Unloaded.")
end

function ESP.Toggle()
    Config.Enabled = not Config.Enabled
    if not Config.Enabled then
        restoreHitboxes()
    end
    print("[ESP/Aimbot] Status changed to: " .. (Config.Enabled and "ON" or "OFF"))
end

_G.UltimateESP = ESP

print("[ESP/Aimbot] PREMIUM EDITION loaded successfully!")
print("Use: _G.UltimateESP.Toggle() to Toggle | _G.UltimateESP.Unload() to Unload.")

return ESP
